# Mugger ID Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A static GitHub Pages site where a Torn player pastes their own API key and sees likely suspects for the stealthed mugs against them, with a fight-log drill-down.

**Architecture:** Plain ES-module JavaScript, no build step, no libraries. Pure modules (`parse.js`, `analyze.js`) do all reasoning and are unit-tested with `node --test`; `torn.js` talks to the Torn API with an injectable `fetch`; `cache.js` stores immutable attack logs in IndexedDB; `ui.js`/`app.js`/`index.html` render the page. Everything runs in the browser; the key only goes to `api.torn.com`.

**Tech Stack:** HTML, CSS, JavaScript (ES2022 modules), IndexedDB, Node 24 `node:test` for tests, GitHub Pages.

**Spec:** `docs/superpowers/specs/2026-10-22-mugger-id-design.md`

## Global Constraints

- No build step, no npm dependencies, no external scripts/styles/fonts. Only network destination: `https://api.torn.com/v2`.
- API key sent only as header `Authorization: ApiKey <key>`, never in a URL.
- Key storage notice text, verbatim: *"Your API key is stored only on your own device (and only if you tick 'remember'). It is never sent to or stored by this site — it goes straight to api.torn.com."*
- Every suspect display carries "probable, not proof".
- All API-provided strings rendered via `textContent` / text nodes — never `innerHTML`.
- **No real player names, IDs, faction names or attack codes anywhere in the repo** (code, tests, docs, commit messages). Before every commit run the denylist check:
  `grep -rniF -f ../.mugger-id-denylist.txt --exclude-dir=.git . && echo "REAL DATA FOUND" || echo clean` — must print `clean`. (The denylist lives outside the repo on purpose.)
- Pace API calls at ≤ 80/min (750 ms apart). Torn timestamps are UTC (TCT).
- Page must work at ~400 px width and in light and dark themes.
- Commit messages end with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

```js
// Turns Torn attack-log entries into the attacker's side of the fight.
// Stealthed attackers appear as "Someone", but their weapons and damage stay in the text.

const WEAPON_RE = /\b(?:their|his|her) ([A-Z].*?)(?: in the | hitting | missing |$)/;

export function classifyLine(text, { attackerName, defenderName }) {
  const t = text.toLowerCase();
  const startsWith = (name) => {
    if (!name) return false;
    const n = name.toLowerCase();
    return t === n || t.startsWith(`${n} `) || t.startsWith(`${n}'`);
  };
  if (startsWith(defenderName)) return 'defender';
  // Visible fight: only the named attacker's own lines count. "Someone" lines there belong to
  // stealthed joiners or to a different fight that ended up in the same log.
  if (attackerName) return startsWith(attackerName) ? 'attacker' : 'other';
  return t.startsWith('someone ') ? 'attacker' : 'other';
}

export function weaponOf(text) {
  if (text.includes(' kicked ')) return 'Kick';
  const thrown = /threw an? (.+?) (?:at|near) /.exec(text);
  if (thrown) return thrown[1];
  const m = WEAPON_RE.exec(text);
  return m ? m[1].replace(/ critically$/, '').trim() : null;
}

const toNumber = (s) => Number(s.replace(/,/g, ''));

export function parseLog(entries, names) {
  const weapons = {};
  const hits = [];
  let mugAmount = null;
  for (const e of entries) {
    if (classifyLine(e.text, names) !== 'attacker') continue;
    if (e.action === 'mug') {
      const m = /\$([\d,]+)/.exec(e.text);
      mugAmount = m ? toNumber(m[1]) : 0;
      continue;
    }
    const weapon = weaponOf(e.text);
    if (!weapon) continue;
    weapons[weapon] = (weapons[weapon] || 0) + 1;
    const dmg = / for ([\d,]+)$/.exec(e.text);
    if (dmg) {
      const part = / in the ([A-Za-z ]+?) for /.exec(e.text);
      hits.push({ weapon, damage: toNumber(dmg[1]), crit: e.text.includes('critically'), part: part ? part[1] : null });
    }
  }
  return { weapons, hits, mugAmount };
}
```

```js
// Fingerprints fights and scores visible attackers as suspects for stealthed mugs.
// Pure functions: no network, no DOM.
import { parseLog } from './parse.js';

export const DAY = 86400;
const WINDOW = 14 * DAY;
const FIST_WEAPONS = new Set(['Fists', 'Kick']);
export const RANK = { High: 3, Medium: 2, Low: 1 };

export function fingerprint(attack, attacklog) {
  const names = { attackerName: attack.attacker?.name ?? null, defenderName: attack.defender?.name ?? null };
  const parsed = attacklog ? parseLog(attacklog.log, names) : { weapons: {}, hits: [], mugAmount: null };
  return {
    id: attack.id,
    code: attack.code,
    started: attack.started,
    result: attack.result,
    stealthed: attack.attacker == null,
    attacker: attack.attacker
      ? { id: attack.attacker.id, name: attack.attacker.name, faction: attack.attacker.faction?.name ?? null }
      : null,
    ff: attack.modifiers?.fair_fight ?? null,
    overseas: (attack.modifiers?.overseas ?? 1) > 1,
    respect: attack.respect_gain ?? 0,
    weapons: parsed.weapons,
    hits: parsed.hits,
    mugAmount: parsed.mugAmount,
    logMissing: !attacklog,
  };
}

export function buildCandidates(fps) {
  const byId = new Map();
  for (const fp of fps) {
    if (fp.stealthed) continue;
    let c = byId.get(fp.attacker.id);
    if (!c) {
      c = { id: fp.attacker.id, name: fp.attacker.name, faction: fp.attacker.faction, lastSeen: 0, fights: [], weapons: {} };
      byId.set(c.id, c);
    }
    c.fights.push(fp);
    if (fp.started >= c.lastSeen) {
      c.lastSeen = fp.started;
      c.faction = fp.attacker.faction;
    }
    for (const [w, n] of Object.entries(fp.weapons)) c.weapons[w] = (c.weapons[w] || 0) + n;
  }
  return [...byId.values()];
}

const realWeapons = (fp) => Object.keys(fp.weapons).filter((w) => !FIST_WEAPONS.has(w));
const gapTo = (mug) => (f) => Math.abs(f.started - mug.started);

// How many candidates active around the time of this mug used each weapon. Rarer = more telling.
export function weaponWeights(mug, candidates) {
  const users = {};
  for (const c of candidates) {
    if (!c.fights.some((f) => gapTo(mug)(f) <= WINDOW)) continue;
    for (const w of Object.keys(c.weapons)) users[w] = (users[w] || 0) + 1;
  }
  const weights = {};
  for (const [w, n] of Object.entries(users)) weights[w] = FIST_WEAPONS.has(w) ? 0 : 1 / n;
  return weights;
}

function median(xs) {
  const s = [...xs].sort((a, b) => a - b);
  const m = s.length >> 1;
  return s.length % 2 ? s[m] : (s[m - 1] + s[m]) / 2;
}

const nonCrit = (hits, weapon) => hits.filter((h) => h.weapon === weapon && !h.crit && h.damage > 0).map((h) => h.damage);

function describeGap(sec) {
  const d = Math.round(sec / DAY);
  return d === 0 ? 'same day' : `${d} day${d === 1 ? '' : 's'} away`;
}

export function confidenceOf(s, hasWeapon) {
  if (s.faction === 'strongAgainst') return null;
  const weaponMatch = s.weapon === 'strong' || s.weapon === 'weak';
  const ffMatch = s.ff === 'strong' || s.ff === 'moderate';
  const timeOk = s.time !== 'none';
  const negatives = (s.weapon === 'against') + (s.ff === 'against') + (s.faction === 'weakAgainst');
  let level = null;
  if (s.weapon === 'strong' && ffMatch && s.time === 'strong' && negatives === 0) level = 'High';
  else if ([weaponMatch, ffMatch, timeOk].filter(Boolean).length >= 2 && negatives === 0) level = 'Medium';
  else if ((weaponMatch || ffMatch) && s.weapon !== 'against') level = 'Low';
  if (!hasWeapon && level) level = 'Low'; // fists/kicks only: too easy to match by coincidence
  return level;
}

function scoreOf(s) {
  const pts = { strong: 2, moderate: 1, weak: 1 };
  return (pts[s.weapon] || 0) + (pts[s.ff] || 0) + (pts[s.time] || 0) + (s.damage ? 1 : 0) + (s.overseas ? 1 : 0);
}

export function scoreCandidate(mug, cand, weights) {
  const reasons = [];
  const signals = { weapon: 'none', ff: 'none', time: 'none', faction: 'ok', damage: false, overseas: false };
  const gap = gapTo(mug);

  const sWeapons = realWeapons(mug);
  if (sWeapons.length) {
    const shared = sWeapons.filter((w) => cand.weapons[w]);
    if (shared.length) {
      const best = shared.reduce((a, b) => ((weights[b] ?? 0) > (weights[a] ?? 0) ? b : a));
      signals.weapon = (weights[best] ?? 0) >= 0.5 ? 'strong' : 'weak';
      const n = cand.fights.filter((f) => f.weapons[best]).length;
      reasons.push(`${best} used in ${n} visible fight${n === 1 ? '' : 's'}${signals.weapon === 'weak' ? ' (common weapon)' : ''}`);
    } else {
      signals.weapon = 'against';
      reasons.push(`never seen using ${sWeapons.join(', ')}`);
    }
  }

  const near = cand.fights.filter((f) => f.ff != null && gap(f) <= WINDOW);
  if (mug.ff != null && near.length) {
    const closest = near.reduce((a, b) => (Math.abs(b.ff - mug.ff) < Math.abs(a.ff - mug.ff) ? b : a));
    const diff = Math.abs(closest.ff - mug.ff);
    let level = diff <= 0.03 ? 'strong' : diff <= 0.1 ? 'moderate' : diff > 0.3 ? 'against' : 'none';
    const edge = mug.ff === 3 || mug.ff === 1;
    if (edge && level === 'strong') level = 'moderate';
    else if (edge && level === 'moderate') level = 'none';
    signals.ff = level;
    if (level !== 'none') {
      const note = edge ? (mug.ff === 3 ? ' (cap, weaker)' : ' (floor, weaker)') : '';
      reasons.push(`FF ${mug.ff.toFixed(2)} vs ${closest.ff.toFixed(2)}${note}${level === 'against' ? ' — too far apart' : ''}`);
    }
  }

  const minGap = Math.min(...cand.fights.map(gap));
  if (minGap <= 3 * DAY) signals.time = 'strong';
  else if (minGap <= WINDOW) signals.time = 'moderate';
  if (signals.time !== 'none') reasons.push(`visible hit ${describeGap(minGap)}`);

  const nearest = cand.fights.reduce((a, b) => (gap(b) < gap(a) ? b : a));
  const candInFaction = nearest.attacker.faction != null;
  if (mug.respect > 0 && !candInFaction) {
    signals.faction = 'strongAgainst';
    reasons.push('stealthed attacker earned respect but this player had no faction');
  } else if (mug.respect === 0 && candInFaction) {
    signals.faction = 'weakAgainst';
    reasons.push('no respect earned but this player was in a faction');
  }

  for (const w of sWeapons) {
    const a = nonCrit(mug.hits, w);
    const b = cand.fights.flatMap((f) => nonCrit(f.hits, w));
    if (a.length && b.length) {
      const ma = median(a);
      const mb = median(b);
      if (Math.abs(ma - mb) <= 0.35 * mb) {
        signals.damage = true;
        reasons.push(`${w} damage similar (~${Math.round(ma)} vs ~${Math.round(mb)})`);
        break;
      }
    }
  }

  if (mug.overseas && cand.fights.some((f) => f.overseas && gap(f) <= 3 * DAY)) {
    signals.overseas = true;
    reasons.push('also attacking abroad around then');
  }

  return { signals, confidence: confidenceOf(signals, sWeapons.length > 0), score: scoreOf(signals), reasons };
}
```

```js
function summarize(results) {
  const byConfidence = { High: 0, Medium: 0, Low: 0, None: 0 };
  let total = 0;
  for (const r of results) {
    total += r.mug.mugAmount ?? 0;
    byConfidence[r.suspects[0]?.confidence ?? 'None'] += 1;
  }
  return { count: results.length, total, byConfidence };
}

function buildLeaderboard(results) {
  const rows = new Map();
  for (const { mug, suspects } of results) {
    const top = suspects[0];
    if (!top || top.confidence === 'Low') continue;
    const row = rows.get(top.candidate.id) ?? { ...top.candidate, mugs: 0, total: 0, best: top.confidence };
    row.mugs += 1;
    row.total += mug.mugAmount ?? 0;
    if (RANK[top.confidence] > RANK[row.best]) row.best = top.confidence;
    rows.set(top.candidate.id, row);
  }
  return [...rows.values()].sort((a, b) => RANK[b.best] - RANK[a.best] || b.mugs - a.mugs);
}

export function analyze(attacks, logs) {
  const fingerprints = attacks.map((a) => fingerprint(a, logs.get(a.code)));
  const candidates = buildCandidates(fingerprints);
  const mugs = fingerprints.filter((f) => f.stealthed && f.result === 'Mugged').sort((a, b) => b.started - a.started);
  const results = mugs.map((mug) => {
    const weights = weaponWeights(mug, candidates);
    const suspects = candidates
      .map((c) => ({ candidate: { id: c.id, name: c.name, faction: c.faction }, ...scoreCandidate(mug, c, weights) }))
      .filter((s) => s.confidence)
      .sort((a, b) => RANK[b.confidence] - RANK[a.confidence] || b.score - a.score)
      .slice(0, 3);
    return { mug, suspects };
  });
  return { results, candidates, fingerprints, summary: summarize(results), leaderboard: buildLeaderboard(results) };
}
```

```js
// Torn API v2 client. The key is only ever sent to api.torn.com, in the Authorization header.
const BASE = 'https://api.torn.com/v2';

export const ERROR_MESSAGES = {
  0: 'Torn API returned an unknown error.',
  2: 'That API key is incorrect.',
  5: 'Too many requests. Torn allows 100 per minute per key.',
  8: 'Your IP is temporarily blocked by Torn for too many requests.',
  9: 'The Torn API is temporarily disabled. Try again later.',
  10: 'This key belongs to a player in federal jail.',
  13: 'This key is disabled because its owner has been inactive for more than 7 days.',
  16: "This key's access level is too low. Create a Limited key.",
  18: 'This key is paused by its owner.',
};

export class TornError extends Error {
  constructor(code, message) {
    super(ERROR_MESSAGES[code] ?? message ?? `Torn API error ${code}`);
    this.code = code;
  }
}

export function createClient(key, opts = {}) {
  const fetchFn = opts.fetchFn ?? ((...args) => fetch(...args));
  const sleep = opts.sleep ?? ((ms) => new Promise((resolve) => setTimeout(resolve, ms)));
  const now = opts.now ?? (() => Date.now());
  const minIntervalMs = opts.minIntervalMs ?? 750;
  let last = -Infinity;

  async function get(path) {
    for (let attempt = 0; ; attempt++) {
      const wait = last + minIntervalMs - now();
      if (wait > 0) await sleep(wait);
      last = now();
      let body;
      try {
        const res = await fetchFn(BASE + path, { headers: { Authorization: `ApiKey ${key}` } });
        body = await res.json();
      } catch {
        if (attempt < 2) {
          await sleep(2000 * (attempt + 1));
          continue;
        }
        throw new TornError(-1, 'Could not reach the Torn API. Check your connection.');
      }
      if (body.error) {
        if (body.error.code === 5 && attempt < 3) {
          opts.onRateLimit?.();
          await sleep(60000);
          continue;
        }
        throw new TornError(body.error.code, body.error.error);
      }
      return body;
    }
  }

  async function keyInfo() {
    const access = (await get('/key/info')).info?.access ?? {};
    return { level: access.level ?? 0, type: access.type ?? 'Unknown' };
  }

  async function incomingAttacks(fromTs, onProgress) {
    const seen = new Map();
    let to = Math.floor(now() / 1000);
    for (;;) {
      const page = (await get(`/user/attacks?filters=incoming&sort=DESC&limit=100&from=${fromTs}&to=${to}`)).attacks ?? [];
      for (const a of page) if (a.started >= fromTs) seen.set(a.id, a);
      onProgress?.(seen.size);
      if (page.length < 100) break;
      const oldest = Math.min(...page.map((a) => a.started));
      if (oldest <= fromTs || oldest >= to) break;
      to = oldest;
    }
    return [...seen.values()].sort((a, b) => b.started - a.started);
  }

  const attackLog = async (code) => (await get(`/torn/attacklog?log=${encodeURIComponent(code)}`)).attacklog;
  const profile = async (id) => (await get(`/user/${id}/profile`)).profile;

  return { get, keyInfo, incomingAttacks, attackLog, profile };
}
```

```js
// Attack logs never change, so they are cached on this device by attack code.
// Falls back to memory when IndexedDB is unavailable (private windows, blocked storage).
const DB_NAME = 'mugger-id';
const STORE = 'logs';

function openDb() {
  return new Promise((resolve, reject) => {
    const req = indexedDB.open(DB_NAME, 1);
    req.onupgradeneeded = () => req.result.createObjectStore(STORE);
    req.onsuccess = () => resolve(req.result);
    req.onerror = () => reject(req.error);
  });
}

export async function openCache() {
  const memory = new Map();
  let db = null;
  try {
    db = await openDb();
  } catch {
    db = null;
  }
  const run = (mode, fn) =>
    new Promise((resolve, reject) => {
      const tx = db.transaction(STORE, mode);
      const req = fn(tx.objectStore(STORE));
      tx.oncomplete = () => resolve(req.result);
      tx.onerror = () => reject(tx.error);
    });

  return {
    async get(code) {
      if (memory.has(code)) return memory.get(code);
      if (!db) return undefined;
      try {
        const value = await run('readonly', (s) => s.get(code));
        if (value) memory.set(code, value);
        return value;
      } catch {
        return undefined;
      }
    },
    async put(code, log) {
      memory.set(code, log);
      if (!db) return;
      try {
        await run('readwrite', (s) => s.put(log, code));
      } catch {
        // memory copy is enough for this session
      }
    },
    async clear() {
      memory.clear();
      if (!db) return;
      try {
        await run('readwrite', (s) => s.clear());
      } catch {
        // nothing stored
      }
    },
  };
}
```

```js
  const el = document.createElement(tag);
  for (const [k, v] of Object.entries(attrs)) {
    if (v == null || v === false) continue;

    else el.setAttribute(k, v === true ? '' : v);

export function renderMugList(el, { results }, { onlyStrong, onSelect }) {
  const rows = results.filter((r) => !onlyStrong || ['High', 'Medium'].includes(r.suspects[0]?.confidence));
  if (!rows.length) {
    el.replaceChildren(h('p', { class: 'muted' }, results.length ? 'No mugs match this filter.' : 'No stealthed mugs in this period.'));
    return;
  }
  el.replaceChildren(
    h('div', { class: 'scroll' },
      h('table', { class: 'clickable' },
        h('thead', {}, h('tr', {}, h('th', {}, 'When'), h('th', { class: 'num' }, 'Taken'), h('th', { class: 'num' }, 'FF'), h('th', {}, 'Weapons'), h('th', {}, 'Top suspect'))),

          return h('tr', { tabindex: 0, onclick: open, onkeydown: (e) => { if (e.key === 'Enter') open(); } },

            h('td', {}, weaponsText(r.mug)),

  );

function logList(log) {
  return log ? h('ol', { class: 'log' }, log.log.map((e) => h('li', {}, e.text))) : h('p', { class: 'muted' }, 'Log unavailable.');

export function renderDrilldown(el, row, { logFor, fingerprints, loadProfile, onClose }) {
  const { mug, suspects } = row;

      .catch((e) => profileBox.replaceChildren(`Profile unavailable: ${e.message}`));
    const fights = fingerprints.filter((f) => f.attacker?.id === s.candidate.id).sort((a, b) => b.started - a.started);
    return h('section', { class: 'suspect' },
      h('h4', {}, profileLink(s.candidate), ' ', badge(s.confidence)),

      profileBox,
      h('details', {},
        h('summary', {}, `Their visible fights on you (${fights.length})`),
        h('ul', { class: 'fights' }, fights.map((f) =>
          h('li', {}, h('details', {},
            h('summary', {}, `${when(f.started)} · ${f.result} · FF ${ffText(f.ff)} · ${weaponsText(f)}`),
            logList(logFor(f.code))))))));
  });
  el.replaceChildren(
    h('div', { class: 'drill-head' },
      h('h3', {}, `Stealthed mug · ${when(mug.started)} · ${money(mug.mugAmount)}`),
```

```js
import { createClient, TornError } from './torn.js';
import { openCache } from './cache.js';
import { analyze } from './analyze.js';
import { renderSummary, renderLeaderboard, renderMugList, renderDrilldown } from './ui.js';

const KEY_STORE = 'mugger-id:key';
const $ = (id) => document.getElementById(id);
const storedKey = {
  get() { try { return localStorage.getItem(KEY_STORE); } catch { return null; } },
  set(v) { try { localStorage.setItem(KEY_STORE, v); } catch { /* storage blocked */ } },
  clear() { try { localStorage.removeItem(KEY_STORE); } catch { /* storage blocked */ } },
};

let cancelled = false;
let state = null; // { client, attacks, logs, result }
const profiles = new Map();

function setError(message) {
  $('error').textContent = message ?? '';
  $('error').hidden = !message;
}

function setProgress(label, done = 0, total = 0) {
  $('progress-label').textContent = label;
  if (total) {
    $('progress').max = total;
    $('progress').value = done;
  } else {
    $('progress').removeAttribute('value'); // indeterminate
  }
}

function render() {
  if (!state) return;
  state.result = analyze(state.attacks, state.logs);
  $('results').hidden = false;
  renderSummary($('summary'), state.result);
  renderLeaderboard($('leaderboard'), state.result);
  renderMugList($('mugs'), state.result, { onlyStrong: $('only-strong').checked, onSelect: openDrill });
}

function loadProfile(id) {
  if (!profiles.has(id)) {
    profiles.set(id, state.client.profile(id).catch((e) => { profiles.delete(id); throw e; }));
  }
  return profiles.get(id);
}

function openDrill(row) {
  const el = $('drill');
  el.hidden = false;
  renderDrilldown(el, row, {
    logFor: (code) => state.logs.get(code),
    fingerprints: state.result.fingerprints,
    loadProfile,
    onClose: () => { el.hidden = true; },
  });
  el.scrollIntoView({ behavior: 'smooth', block: 'start' });
}

async function run(key, days) {
  cancelled = false;
  setError(null);
  $('cleared').hidden = true;
  $('drill').hidden = true;
  $('working').hidden = false;
  $('analyze').disabled = true;
  const client = createClient(key, { onRateLimit: () => setProgress('Torn rate limit reached. Waiting 60 seconds…') });
  try {
    setProgress('Checking your key…');
    const info = await client.keyInfo();
    if (info.level !== 0 && info.level < 3) throw new TornError(16);

    const fromTs = Math.floor(Date.now() / 1000) - days * 86400;
    const attacks = await client.incomingAttacks(fromTs, (n) => setProgress(`Loading your incoming attacks… ${n}`));
    const cache = await openCache();
    const wanted = attacks.filter((a) => a.attacker || a.result === 'Mugged');
    const logs = new Map();
    const todo = [];
    for (const a of wanted) {
      const cached = await cache.get(a.code);
      if (cached) logs.set(a.code, cached);
      else todo.push(a);
    }
    state = { client, attacks, logs, result: null };
    render();

    for (let i = 0; i < todo.length && !cancelled; i++) {
      const minutes = Math.ceil(((todo.length - i) * 0.75) / 60);
      setProgress(`Fetching fight logs ${i} / ${todo.length} (about ${minutes} min left)`, i, todo.length);
      try {
        const log = await client.attackLog(todo[i].code);
        logs.set(todo[i].code, log);
        await cache.put(todo[i].code, log);
      } catch (e) {
        if (!(e instanceof TornError) || e.code !== -1) throw e; // network failure: skip this log
      }
      if (i % 20 === 19) render();
    }
    render();
  } catch (e) {
    setError(e.message);
  } finally {
    $('working').hidden = true;
    $('analyze').disabled = false;
  }
}

const saved = storedKey.get();
if (saved) {
  $('key').value = saved;
  $('remember').checked = true;
}

$('start-form').addEventListener('submit', (e) => {
  e.preventDefault();
  const key = $('key').value.trim();
  if (!/^[A-Za-z0-9]{16}$/.test(key)) {
    setError('Torn API keys are 16 letters and numbers.');
    return;
  }
  if ($('remember').checked) storedKey.set(key);
  else storedKey.clear();
  run(key, Number($('days').value));
});
$('remember').addEventListener('change', () => { if (!$('remember').checked) storedKey.clear(); });
$('cancel').addEventListener('click', () => { cancelled = true; });
$('only-strong').addEventListener('change', render);
$('clear').addEventListener('click', async () => {
  await (await openCache()).clear();
  storedKey.clear();
  $('key').value = '';
  $('remember').checked = false;
  setError(null);
  $('cleared').hidden = false;
});
```
